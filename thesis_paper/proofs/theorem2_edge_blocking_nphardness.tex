\section[Proof of Theorem 2: NP-Hardness of Multi-Principal Access Interdiction]{Proof of Theorem 2: NP-Hardness of Multi-Principal Access Interdiction in Enterprise Identity Graphs}
\label{proof:theorem2}

\begin{theorem}[NP-Hardness of Multi-Principal Access Interdiction]
Let $G = (V, E)$ be a directed multigraph representing an Active Directory domain, where $V$ represents security principals and directory objects, and $E$ represents directed authorization, delegation, and administrative permissions. Let $c: E \to \mathbb{R}^+$ assign operational disruption costs to edges, representing the business friction of revoking a specific privilege.

Let $\mathcal{P}_{\text{forbidden}} = \{(s_1, t_1), (s_2, t_2), \dots, (s_k, t_k)\}$ denote a set of $k \ge 3$ forbidden source-target compromise pairs (e.g., specific unprivileged footholds and their respective target administrative assets across distinct trust or tier boundaries) that must be severed.

The \textbf{Multi-Principal Access Interdiction (MPAI)} problem asks: given an operational disruption budget $K \in \mathbb{R}^+$, does there exist an edge interdiction subset $E_{\text{cut}} \subseteq E$ such that:
\begin{enumerate}
    \item Total operational revocation cost is bounded:
    \begin{equation}
    \sum_{e \in E_{\text{cut}}} c(e) \le K
    \end{equation}
    \item All forbidden directed compromise paths are severed:
    \begin{equation}
    \forall (s_i, t_i) \in \mathcal{P}_{\text{forbidden}}, \quad \text{Path}(s_i \rightsquigarrow t_i) \text{ does not exist in } G' = (V, E \setminus E_{\text{cut}})
    \end{equation}
\end{enumerate}
The MPAI problem is NP-complete for $k \ge 3$. Consequently, finding the optimal edge-severing defense strategy that minimizes operational disruption while neutralizing multiple independent lateral compromise paths is NP-hard.
\end{theorem}

\begin{proof}
We prove NP-completeness by establishing:
\begin{enumerate}
    \item $\text{MPAI} \in \text{NP}$, and
    \item A polynomial-time reduction from the classical NP-complete \textbf{Directed Multiway Cut} problem~\cite{garg1994multiway,dahlstrm2000multiway} to MPAI:
    \begin{equation}
    \text{Directed Multiway Cut} \le_p \text{MPAI}
    \end{equation}
\end{enumerate}

\paragraph{Part 1: Membership in NP.}
Given a candidate edge subset $E_{\text{cut}} \subseteq E$:
\begin{enumerate}
    \item We compute the total operational revocation cost $\sum_{e \in E_{\text{cut}}} c(e)$ in $\mathcal{O}(|E_{\text{cut}}|)$ time and verify whether $\sum_{e \in E_{\text{cut}}} c(e) \le K$.
    \item We construct the residual graph $G' = (V, E \setminus E_{\text{cut}})$ in $\mathcal{O}(|V| + |E|)$ time.
    \item For each of the $k$ forbidden pairs $(s_i, t_i) \in \mathcal{P}_{\text{forbidden}}$, we execute Breadth-First Search (BFS) starting from $s_i$ in $G'$ to verify that $t_i$ is unreachable. This requires $\mathcal{O}(k \cdot (|V| + |E|))$ time.
\end{enumerate}
Because all verification steps complete in deterministic polynomial time with respect to the graph size, $\text{MPAI} \in \text{NP}$.

\paragraph{Part 2: Polynomial-Time Reduction from Directed Multiway Cut.}
Recall the definition of the \textbf{Directed Multiway Cut} problem, proven NP-complete by Garg, Vazirani, and Yannakakis~\cite{garg1994multiway}:
\begin{quote}
\textbf{Directed Multiway Cut:} Given a directed graph $H = (V_H, E_H)$ with positive edge weights $w: E_H \to \mathbb{R}^+$, a subset of $k \ge 3$ distinct terminal vertices $X = \{x_1, x_2, \dots, x_k\} \subseteq V_H$, and a cost threshold $W$, does there exist an edge subset $C \subseteq E_H$ with $\sum_{e \in C} w(e) \le W$ such that no directed path connects any terminal $x_i$ to any other terminal $x_j$ ($i \neq j$) in the residual graph $H' = (V_H, E_H \setminus C)$?
\end{quote}

Given an arbitrary instance $\langle H = (V_H, E_H), w, X = \{x_1, \dots, x_k\}, W \rangle$ of the Directed Multiway Cut problem ($k \ge 3$), we construct an instance $\langle G = (V, E), c, \mathcal{P}_{\text{forbidden}}, K \rangle$ of MPAI in polynomial time:

\begin{enumerate}
    \item \textbf{Vertex Construction:} For each terminal $x_i \in X$, create two distinct nodes in $G$: an attacker source principal $s_i$ and a high-value target asset $t_i$. For all non-terminal vertices $v \in V_H \setminus X$, include $v \in V$. Thus:
    \begin{equation}
    V = (V_H \setminus X) \cup \{s_1, \dots, s_k\} \cup \{t_1, \dots, t_k\}
    \end{equation}

    \item \textbf{Edge and Cost Construction:}
    \begin{enumerate}
        \item For every original edge $(u, v) \in E_H$:
        \begin{enumerate}
            \item If $u \notin X$ and $v \notin X$, add $(u, v) \in E$ with cost $c(u, v) = w(u, v)$.
            \item If $u = x_i \in X$ and $v \notin X$, add $(s_i, v) \in E$ with cost $c(s_i, v) = w(x_i, v)$.
            \item If $u \notin X$ and $v = x_j \in X$, add $(u, t_j) \in E$ with cost $c(u, t_j) = w(u, x_j)$.
            \item If $u = x_i \in X$ and $v = x_j \in X$ ($i \neq j$), add $(s_i, t_j) \in E$ with cost $c(s_i, t_j) = w(x_i, x_j)$.
        \end{enumerate}
        \item For each terminal index $i \in \{1, \dots, k\}$, add a directed continuation edge $(t_i, s_i)$ with infinite capacity $c(t_i, s_i) = \infty$ (or $W + 1$). This enforces that reaching target $t_i$ allows traversing any path leaving source $s_i$, preserving the full reachability topology of vertex $x_i$ in $H$.
    \end{enumerate}

    \item \textbf{Forbidden Pair Definition:} Set the forbidden compromise pair set to all distinct terminal combinations:
    \begin{equation}
    \mathcal{P}_{\text{forbidden}} = \{(s_i, t_j) \mid 1 \le i, j \le k, \, i \neq j\}
    \end{equation}
    Here $|\mathcal{P}_{\text{forbidden}}| = k(k-1) \ge 6$ for $k \ge 3$.

    \item \textbf{Disruption Budget:} Set $K = W$.
\end{enumerate}

The reduction generates $|V| = |V_H| + k$ vertices and $|E| = |E_H| + k$ edges, requiring $\mathcal{O}(|V_H| + |E_H|)$ time.

\paragraph{Equivalence of Solutions:}
\begin{enumerate}
    \item $(\Rightarrow)$ Suppose there exists a valid Directed Multiway Cut $C \subseteq E_H$ in $H$ with $\sum_{e \in C} w(e) \le W$ that disconnects all pairs $x_i \rightsquigarrow x_j$ for $i \neq j$. Let $E_{\text{cut}} \subseteq E$ be the identical set of edges in $G$. Since $C$ contains no infinite-cost edges, $\sum_{e \in E_{\text{cut}}} c(e) = \sum_{e \in C} w(e) \le W = K$. Furthermore, because no path connects $x_i$ to $x_j$ in $H \setminus C$, no path can connect $s_i$ to $t_j$ in $G \setminus E_{\text{cut}}$. Hence, all pairs in $\mathcal{P}_{\text{forbidden}}$ are severed.

    \item $(\Leftarrow)$ Conversely, suppose there exists an edge cut $E_{\text{cut}} \subseteq E$ in $G$ with cost $\sum_{e \in E_{\text{cut}}} c(e) \le K = W$ severing all paths for every pair $(s_i, t_j) \in \mathcal{P}_{\text{forbidden}}$ ($i \neq j$). Because $K < \infty$, $E_{\text{cut}}$ cannot contain any infinite-capacity continuation edge $(t_i, s_i)$. Thus, $E_{\text{cut}}$ corresponds directly to a valid edge subset $C \subseteq E_H$ in $H$. If there existed a directed path from $x_i$ to $x_j$ ($i \neq j$) in $H \setminus C$, the identical sequence of edges would form a directed path from $s_i$ to $t_j$ in $G \setminus E_{\text{cut}}$, contradicting the assumption that $(s_i, t_j)$ is severed. Therefore, $C$ separates all terminal pairs in $H$ with cost $\le W$.
\end{enumerate}

This establishes that MPAI is NP-complete for $k \ge 3$. The optimization problem of finding the minimal-disruption edge-severing set is therefore NP-hard.
\end{proof}
